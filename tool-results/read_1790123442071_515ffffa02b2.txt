    13→// ---------- Tipe payload API (minimal, client-side) ----------
    14→
    15→interface Acct { id: string; kind: string; name: string; balance: number }
    16→interface Org { id: string; code: string; kind: string; name: string; lifecycle: string; specialization: string | null; accounts: Acct[] }
    17→interface Agent { id: string; code: string; name: string; role: string; orgCode: string; orgKind: string; status: string; reputation: number; grants: number }
    18→interface Proposal { id: string; kind: string; orgCode: string | null; institution: string; status: string; payload: Record<string, unknown>; policyCheck: Record<string, unknown>; reason: string; decidedBy: string | null; createdAt: string }
    19→interface Task { id: string; agentCode: string; orgCode: string; type: string; status: string; route: Record<string, unknown>; result: Record<string, unknown>; createdAt: string }
    20→interface EventRow { seq: number; type: string; subjectType: string; subjectId: string; payload: Record<string, unknown>; createdAt: string }
    21→interface LedgerRow { id: string; txType: string; purpose: string; isExternal: boolean; counterparty: string | null; amount: number; legs: { side: string; amount: number; account: string; kind: string }[]; createdAt: string }
    22→interface Metrics {
    23→  moneySupply: number; treasury: number; reserveMin: number; govCash: number; taxCollected: number;
    24→  internalTradeVolume: number; externalRevenue: number; externalRevenueReal: number; externalRevenueSandbox: number; expenses: number;
    25→  companies: { code: string; name: string; lifecycle: string; operating: number; income: number; expense: number }[];
    26→  runwayDays: number | null; concentrationTop: number; alerts: string[];
    27→}
    28→interface StatePayload {
    29→  orgs: Org[]; agents: Agent[]; proposals: Proposal[]; tasks: Task[]; events: EventRow[]; ledger: LedgerRow[];
    30→  policies: { key: string; category: string; value: unknown; note: string }[];
    31→  metrics: Metrics;
    32→  ticks: { tick: number; target: string; summary: string; mode: string; model?: string; at: string }[];
    33→  entities: { mcType: string; mcName: string; civCode: string; mcCoords: Record<string, unknown> }[];
    34→  grants: { agentCode: string; capability: string; budgetCap: number }[];
    35→  memories: { id: string; ownerType: string; scope: string; visibility: string; content: string; createdAt: string }[];
    36→  village: Village;
    37→  lastTick: Record<string, unknown> | null;
    38→  counts: { orgs: number; agents: number; events: number; txns: number; memories: number; tasks: number };
    39→  serverTime: string;
    40→}
    41→
    42→interface VillagerRow {
    43→  code: string; name: string; profession: string; profKey: string; role: string;
    44→  source: string; embodiment: string; village: string; mood: string;
    45→  xp: number; socialScore: number; wallet: number; workOrg: string | null;
    46→  lastAction: string | null; lastActionAt: string | null; coords: string | null;
    47→}
    48→interface Village {
    49→  population: number;
    50→  byProfession: Record<string, number>;
    51→  bySource: { SIMULASI: number; CENSUS: number };
    52→  byEmbodiment: { EMBODIED: number; DREAMING: number; MISSING: number };
    53→  retired: number;
    54→  villagers: VillagerRow[];
    55→  wagesPaid: number;
    56→  directives?: DirectiveBlock;
    57→  market?: MarketBlock;
    58→}
    59→
    60→// SLICE 8 — tubuh warga (direktif) + pasar desa
    61→interface DirectiveRow { id: string; villagerCode: string; villagerName: string; kind: string; status: string; origin: string; payload: { to?: { x: number; y: number; z: number }; message?: string; orgCode?: string }; result: string | null; createdAt: string }
    62→interface DirectiveBlock { queued: number; dispatched: number; applied: number; failed: number; expired: number; recent: DirectiveRow[] }
    63→interface MarketOffer { id: string; seller: string; item: string; unitPrice: number; unitPriceLabel: string; qtyAvailable: number; qtySold: number; status: string; createdAt: string }
    64→interface MarketBlock { open: number; closed: number; totalUnitsSold: number; offers: MarketOffer[] }

    92→export function PeradabanView() {
    93→  const [state, setState] = useState<StatePayload | null>(null);
    94→  const [tab, setTab] = useState<(typeof SUBTABS)[number]["key"]>("peta");
    95→  const [busy, setBusy] = useState<string | null>(null);
    96→  const [pulseAt, setPulseAt] = useState(0);
    97→  const [newName, setNewName] = useState("");
    98→  const [newSpec, setNewSpec] = useState("");
    99→  const pulseRef = useRef(0);
   100→
   101→  const load = useCallback(async () => {
   102→    try {
   103→      const res = await fetch("/api/civos/state", { cache: "no-store" });
   104→      const j = (await res.json()) as { ok: boolean; state?: StatePayload };
   105→      if (j.ok && j.state) setState(j.state);
   106→    } catch { /* biarkan polling berikutnya */ }
   107→  }, []);
   108→
   109→  const beat = useCallback(async (target?: string) => {
   110→    setBusy("denyut");
   111→    try {
   112→      const res = await fetch("/api/civos/heartbeat", {
   113→        method: "POST",
   114→        headers: { "Content-Type": "application/json" },
   115→        body: JSON.stringify(target ? { target } : {}),
   116→      });
   117→      const j = (await res.json()) as { ok: boolean; summary?: { at: string } };
   118→      if (j.ok) {
   119→        pulseRef.current = Date.now();
   120→        setPulseAt(pulseRef.current);
   121→      }
   122→      await load();
   123→    } finally {
   124→      setBusy(null);
   125→    }
   126→  }, [load]);
   127→
   128→  const action = useCallback(async (act: string, params?: Record<string, unknown>) => {
   129→    setBusy(act);
   130→    try {
   131→      await fetch("/api/civos/action", {
   132→        method: "POST",
   133→        headers: { "Content-Type": "application/json" },
   134→        body: JSON.stringify({ action: act, params }),
   135→      });
   136→      await load();
   137→    } finally {
   138→      setBusy(null);
   139→    }
   140→  }, [load]);
   141→
   142→  useEffect(() => { void load(); }, [load]);
   143→  useEffect(() => {
   144→    const t = setInterval(() => { void load(); }, 12_000);
   145→    return () => clearInterval(t);
   146→  }, [load]);
   147→  useEffect(() => {
   148→    const t = setInterval(() => { void beat(); }, 50_000);
   149→    return () => clearInterval(t);
   150→  }, [beat]);
   151→
   152→  if (!state) {
   153→    return (
   154→      <div className="p-6 space-y-3">
   155→        <div className="catalog !text-[10px] text-[#fbbf24]">MENGHIDUPKAN PERADABAN…</div>
   156→        <div className="h-40 rounded-md border border-border bg-[#070d0a] animate-pulse" />
   157→      </div>
   158→    );
   159→  }
   160→
   161→  const m = state.metrics;
   162→  const nation = state.orgs.find((o) => o.kind === "NATION");
   163→  const gov = state.orgs.find((o) => o.kind === "GOVERNMENT");
   164→  const companies = state.orgs.filter((o) => o.kind === "COMPANY");
   165→  const city = state.orgs.find((o) => o.kind === "CITY");
   166→  const mode = String(state.lastTick?.mode ?? "INSTITUSI");
   167→  const model = String(state.lastTick?.model ?? "prosedur");
   168→
   169→  const mapEntities: MapEntity[] = state.entities.map((e) => ({
   170→    mcType: e.mcType,
   171→    mcName: e.mcName,
   172→    civCode: e.civCode,
   173→    coords: { x: Number(e.mcCoords.x ?? 0), z: Number(e.mcCoords.z ?? 0) },
   174→  }));
   175→
   176→  return (
   177→    <div className="p-4 md:p-6 space-y-4">
   178→      {/* Kepala peradaban */}
   179→      <header className="rounded-md border border-border bg-[#070d0a] p-4">
   180→        <div className="flex flex-col lg:flex-row lg:items-start gap-4">
   181→          <div className="flex-1 min-w-0">
   182→            <div className="catalog !text-[9px] text-[#fbbf24]">CIVITAS OS — AUTONOMOUS CIVILIZATION OPERATING SYSTEM</div>
   183→            <h1 className="text-xl md:text-2xl font-mono text-[#bfe8cc] mt-1 glow-phos">{nation?.name ?? "Nusantara Digital"}</h1>
   184→            <p className="catalog mt-1 leading-relaxed">
   185→              1 bangsa · 1 pemerintah (4 institusi) · 1 kota · {companies.length} perusahaan · {state.village.population} warga desa (villager otonom) · {state.counts.agents} agen institusi · kernel SQLite otoritatif · dunia Minecraft: mulkymalikuldhr.aternos.me:19132
   186→            </p>
   187→            <div className="flex flex-wrap gap-2 mt-3">
   188→              <span className="catalog border border-[#4ade80]/40 px-2 py-0.5 rounded">UANG BEREDAR {fmt(m.moneySupply)}</span>
   189→              <span className="catalog border border-[#4ade80]/40 px-2 py-0.5 rounded">KAS BANGSA {fmt(m.treasury)}</span>
   190→              <span className="catalog border border-[#fbbf24]/40 px-2 py-0.5 rounded">PAJAK {fmt(m.taxCollected)}</span>
   191→              <span className={`catalog border px-2 py-0.5 rounded ${m.externalRevenueReal === 0 ? "border-[#f87171]/50 text-[#f87171]" : "border-[#4ade80]/50 text-[#4ade80]"}`}>
   192→                REVENUE EKSTERNAL RIIL {fmt(m.externalRevenueReal)} — {m.externalRevenueReal === 0 ? "PRE_REVENUE (JUJUR)" : "AKTIF"}
   193→              </span>
   194→              {m.externalRevenueSandbox > 0 && (

   196→              )}
   197→            </div>
   198→          </div>
   199→          <div className="shrink-0 lg:w-72 space-y-2">
   200→            <div className="rounded border border-border p-3">
   201→              <div className="catalog !text-[9px] text-[#7f9a89]">DENYUT TERAKHIR</div>
   202→              <div className="font-mono text-sm text-[#bfe8cc] mt-1">
   203→                {String(state.lastTick?.target ?? "-")} <span className={`catalog ${mode === "LLM" ? "text-[#4ade80]" : mode === "REFLEX" ? "text-[#fbbf24]" : "text-[#38bdf8]"}`}>[{mode}{mode !== "INSTITUSI" ? `·${model}` : ""}]</span>
   204→              </div>

   256→                </div>

   261→          </div>
   262→        </section>

   265→      {/* DESA — VILLAGER ASCENSION (Slice 7) */}

   268→          {/* Banner kejujuran sumber populasi */}

   270→            <div className="catalog leading-relaxed">
   271→              {state.village.bySource.CENSUS > 0

   273→                : "POPULASI SIMULASI (JUJUR) — server Minecraft sedang tidur, jadi populasi uji berlabel SIMULASI menghidupkan pipeline desa. Begitu bot masuk dunia & mengobservasi villager nyata, sensus CENSUS otomatis menggantikan populasi simulasi (warga SIM mundur, sejarah tetap auditable)."}
   274→            </div>
