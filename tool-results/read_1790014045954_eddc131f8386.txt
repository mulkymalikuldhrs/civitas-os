   313→  applyDecision: async (apply) => {

   367→    if (apply.veto) eventBus.emit("reflect", `VETO konstitusi atas ${c.name}: ${apply.veto}`, { creatureId: c.id, mode: apply.mode });

   369→    // INGAT: ledger keputusan LOKAL (server tetap amnesia). Kontrak koleksi "decisions"

   382→        },

   388→    }
   389→    // Episode berlabel untuk review memori (organs/memory).

   391→      await rememberEpisode(c.id, `[${apply.error ? "FAILED" : "SUCCESS"}:${c.role}] ${apply.phases.putuskan}`.slice(0, 500), apply.error ? "warn" : "info");
   392→    } catch {
   393→      /* abaikan — ledger episode opsional */
   394→    }
   395→    await setSetting("organism.creatures", nextCreatures);
